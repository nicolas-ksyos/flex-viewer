## B. `src/server/seedWriter.ts` — handle `parameters` replacement

In `processStepObject`, add special handling for the `parameters` key before the existing field loop:

When `change.fields.parameters !== undefined`:
1. Find the `parameters` property in the step's object literal
2. If it exists: replace its entire value (the `ObjectLiteralExpression` or `NullKeyword`) with the new stringified value
3. If it doesn't exist: insert it as a new property

The new value is generated by `formatParametersValue(parameters)`:

```ts
function formatParametersValue(value: Record<string, unknown> | null): string {
  if (value === null) return 'null';
  // Pretty-print as a TypeScript object literal (not JSON — use single-quoted strings, no quotes on keys)
  // For simplicity, use JSON.stringify and convert double-quoted keys/strings to a reasonable format
  // Actually: just format as a multi-line object
  return formatObjectLiteral(value, 2);
}

function formatObjectLiteral(obj: Record<string, unknown> | null, depth: number): string {
  if (obj === null) return 'null';
  const indent = '    '.repeat(depth);
  const innerIndent = '    '.repeat(depth + 1);
  const entries = Object.entries(obj).map(([k, v]) => {
    return `${innerIndent}${k}: ${formatValue(v, depth + 1)}`;
  });
  if (entries.length === 0) return '{}';
  return `{\n${entries.join(',\n')},\n${indent}}`;
}

function formatValue(v: unknown, depth: number): string {
  if (v === null) return 'null';
  if (typeof v === 'string') return `'${v.replace(/\\/g, '\\\\').replace(/'/g, "\\'")}'`;
  if (typeof v === 'number') return String(v);
  if (typeof v === 'boolean') return v ? 'true' : 'false';
  if (Array.isArray(v)) {
    if (v.length === 0) return '[]';
    const indent = '    '.repeat(depth);
    const items = v.map(item => '    '.repeat(depth + 1) + formatValue(item, depth + 1));
    return `[\n${items.join(',\n')},\n${indent}]`;
  }
  if (typeof v === 'object' && v !== null) {
    return formatObjectLiteral(v as Record<string, unknown>, depth);
  }
  return JSON.stringify(v);
}
```

The replacement approach: find the `parameters` property assignment's initializer and replace it with the formatted string. Treat it as a single `Replacement { start, end, newText }`.

Handle it BEFORE the main field loop (skip `parameters` in the field loop since it's handled separately).

Also update `getOriginalValue` in `useEditMode.ts` to skip `parameters` field in the auto-prune logic (or just not add it to the prune function — parameters comparison would be complex so skip it entirely):
```ts
case 'parameters': return undefined; // never auto-prune parameters
```
